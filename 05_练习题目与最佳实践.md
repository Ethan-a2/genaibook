class BatchProcessor:
    def __init__(self, batch_size: int = 4, max_wait_time: float = 1.0):
        self.batch_size = batch_size
        self.max_wait_time = max_wait_time
        self.request_queue = []
        self.processing = False
    
    async def add_request(self, request):
        """添加处理请求"""
        self.request_queue.append(request)
        
        # 如果队列满了或等待时间到了，开始处理
        if (len(self.request_queue) >= self.batch_size or 
            self._wait_time_exceeded()):
            await self._process_batch()
    
    async def _process_batch(self):
        """处理一批请求"""
        if not self.request_queue or self.processing:
            return
        
        self.processing = True
        batch = self.request_queue[:self.batch_size]
        self.request_queue = self.request_queue[self.batch_size:]
        
        # 批处理逻辑
        results = await self._process_requests(batch)
        
        # 返回结果
        for request, result in zip(batch, results):
            request.callback(result)
        
        self.processing = False

from functools import lru_cache
import hashlib

class GenerationCache:
    def __init__(self, max_size: int = 1000):
        self.cache = {}
        self.max_size = max_size
        self.access_count = {}
    
    def _generate_key(self, **kwargs) -> str:
        """生成缓存键"""
        # 排序参数以确保一致性
        sorted_kwargs = sorted(kwargs.items())
        param_str = str(sorted_kwargs)
        return hashlib.md5(param_str.encode()).hexdigest()
    
    def get(self, **kwargs) -> Optional[Any]:
        """获取缓存结果"""
        key = self._generate_key(**kwargs)
        if key in self.cache:
            self.access_count[key] = self.access_count.get(key, 0) + 1
            return self.cache[key]
        return None
